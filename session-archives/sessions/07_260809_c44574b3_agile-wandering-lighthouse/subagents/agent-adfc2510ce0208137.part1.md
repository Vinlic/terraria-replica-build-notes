322	    this.blizzardSoundVolume = WeatherState.moveTowards(
323	      this.blizzardSoundVolume, blizzard ? 1 : 0, 0.01 * dayRate);

343	  /** 金币雨掉落（WorldGen.cs:71811-71858） */
344	  private updateCoinRain(ctx: WeatherCtx) {
345	    if (this.coinRain <= 0) return;
346	    if (!this.raining || this.shouldUseStormMusic) { this.coinRain = 0; return; }
347	    if (this.rng.int(0, 29) !== 0) return;
348	    // 全图随机 x、顶部 5% 高度带（:71818-71823）
349	    let x = this.rng.int(50, ctx.maxTilesX - 51) * 16;
350	    const y = this.rng.int(0, Math.floor(this.maxTilesYRef * 0.05) - 1) * 16;
351	    // 玩家锚定（:71825-71832）：最近玩家活跃且未死 且 1/2 → x = player.X ± 2400，
352	    // 钳 [800, (maxTilesX-50)*16]。★此前整段缺失（金币总在全图随机处掉落）
353	    const p = ctx.playerForCoinRain;
354	    if (p && p.active && !p.dead && this.rng.int(0, 1) === 0) {
355	      x = p.x + this.rng.int(-2400, 2400);
356	    }
357	    x = Math.max(800, Math.min((ctx.maxTilesX - 50) * 16, x));
358	    // 非实心 32×32 门由 Game 落点回调侧判定（此处只出提案）
359	    let type: 71 | 72 | 73 = 71;
360	    let stack = this.rng.int(50, 99);
361	    let worth = stack;
362	    if (this.rng.int(0, 2) === 0) { type = 72; stack = this.rng.int(25, 99); worth = stack * 100; }
363	    if (this.rng.int(0, 8) === 0) { type = 73; stack = 1; worth = 1 * 100 * 100; }
364	    ctx.onCoinRain?.(x, y, type, stack);
365	    this.coinRain -= worth;
366	    if (this.coinRain < 0) this.coinRain = 0;
367	  }
