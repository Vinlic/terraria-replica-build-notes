286	function terrainPass({ store, surface, rng, cfg, noise2D }: GenCtx) {
287	  const { w, h } = store;
288	  const base = Math.floor(h * 0.28);
289	  // 多 octave 地表高度。平坦为主、偶发高峰：
290	  // 大势噪声幂次塑形（|v|^1.8）——噪声近 0 的大片区域被压平成平原，
291	  // 只有噪声较强处才隆起成峰；小起伏幅度减半避免处处尖刺
292	  const o1 = createNoise1Dish(rng, 0.012);
293	  const o2 = createNoise1Dish(rng, 0.05);
294	  const o3 = createNoise1Dish(rng, 0.11);
295	  for (let x = 0; x < w; x++) {
296	    const hill = (x > w * 0.18 && x < w * 0.82) ? 1 : 1.6; // 边缘更陡（近似海崖）
297	    const shape = (v: number, amp: number) => Math.sign(v) * Math.pow(Math.abs(v), 1.8) * amp;
298	    surface[x] = base + (shape(o1(x), 30) + o2(x) * 5 + o3(x) * 2) * hill;
299	  }
300	  // 左右海洋：最外侧 6% 下挖成海盆（岸边浅、世界缘深），后续灌水铺沙
301	  const oceanW = Math.max(24, Math.floor(w * 0.06));
302	  const seaLevel = base + 4; // 海平面（水表高度）
303	  for (let x = 0; x < w; x++) {
304	    const edge = Math.min(x, w - 1 - x);
305	    if (edge >= oceanW) continue;
306	    const t = 1 - edge / oceanW; // 0 岸边 → 1 世界缘
307	    const floorY = base + 2 + t * 16 + Math.sin(x * 0.3) * 1.5;
308	    surface[x] = Math.max(surface[x], Math.max(seaLevel + 2, floorY));
309	  }
310	  const rockLevel = Math.floor(h * 0.33);
311	  const nBlend = createNoise1Dish(rng, 0.05);
312	  for (let x = 0; x < w; x++) {
313	    const sy = Math.floor(surface[x]);
314	    // 每列岩石线带噪声起伏（±8 格），形成犬牙交错的土石过渡带
315	    const rl = rockLevel + nBlend(x) * 8;
316	    for (let y = 0; y < h; y++) {
317	      if (y < sy) continue;
318	      let type: number = T.DIRT;
319	      if (y >= rl) type = T.STONE;
320	      else if (y > rl - 14) {
321	        // 过渡带：噪声混入石块夹层
322	        if (fbm(noise2D, x * 0.09, y * 0.09, 2) > 0.12) type = T.STONE;
323	      }
324	      store.setTileSilent(x, y, type);
325	      // 自然墙：地下土/石区背景
326	      if (y > sy + 3) store.setWallSilent(x, y, type === T.STONE ? W.STONE : W.DIRT);
327	    }
328	  }
329	  // 海洋灌水 + 沙滩铺沙：海盆内从海平面灌满水；盆底与近岸滩涂表层铺沙
330	  for (let x = 0; x < w; x++) {
331	    const edge = Math.min(x, w - 1 - x);
332	    const sy = Math.floor(surface[x]);
333	    if (edge < oceanW) {
334	      for (let y = seaLevel; y < sy; y++) store.setLiquid(x, y, 255, 1);
335	      for (let y = sy; y < sy + 5 && y < store.h; y++) {
336	        const t = store.get(x, y);
337	        if (t === T.DIRT || t === T.GRASS) store.setTileSilent(x, y, T.SAND);
338	      }
339	    } else if (edge < oceanW + 8 && surface[x] >= seaLevel - 1) {
340	      // 近岸滩涂：海平面附近的陆地表面沙化过渡
341	      for (let y = sy; y < sy + 4 && y < store.h; y++) {
342	        const t = store.get(x, y);
343	        if (t === T.DIRT || t === T.GRASS) store.setTileSilent(x, y, T.SAND);
344	      }
345	    }
346	    // 沙漠地下层：海洋区沙层之下换沙岩/硬化沙（5-14 格深处随机夹层）
347	    if (edge < oceanW) {
348	      const SANDSTONE = TILE_BY_KEY['sandstone']!, HARDSAND = TILE_BY_KEY['hardened_sand']!;
349	      for (let y = sy + 5; y < sy + 14 && y < store.h; y++) {
350	        const t = store.get(x, y);
351	        if ((t === T.DIRT || t === T.STONE) && rng.next() < 0.45) {
352	          store.setTileSilent(x, y, y < sy + 9 ? SANDSTONE : HARDSAND);
353	        }
354	      }
355	    }
356	  }
357	  // 雪原带：随机一侧的 10%-24% 区间 —— 地表雪化 + 浅层冰夹层（无树：surfacePass 只在草上种树）
358	  {
359	    const ICE = TILE_BY_KEY['ice']!;
360	    const snowLeft = rng.next() < 0.5;
361	    const xs0 = snowLeft ? Math.floor(w * 0.10) : Math.floor(w * 0.76);
362	    const xs1 = snowLeft ? Math.floor(w * 0.24) : Math.floor(w * 0.90);
363	    for (let x = xs0; x <= xs1; x++) {
364	      const sy = Math.floor(surface[x]);
365	      for (let y = sy; y < sy + 4 && y < store.h; y++) {
366	        const t = store.get(x, y);
367	        if (t === T.GRASS || t === T.DIRT) store.setTileSilent(x, y, T.SNOW);
368	      }
369	      for (let y = sy + 4; y < sy + 12 && y < store.h; y++) {
370	        if (store.get(x, y) === T.DIRT && rng.next() < 0.45) store.setTileSilent(x, y, ICE);
371	      }
372	    }
373	  }
374	  // 世界最底部 3 行基岩（不可破坏 → 用 STONE 代替，破坏规则由硬度统一处理，简化）
375	}

384	function cavePass({ store, surface, noise2D, rng, cfg }: GenCtx) {
385	  const { w, h } = store;
386	  // 1) 噪声挖空（海洋列跳过：保护海床，防止海水灌进洞穴系统）
387	  const caveOceanW = Math.max(24, Math.floor(w * 0.06)) + 2;
388	  for (let x = 0; x < w; x++) {
389	    if (Math.min(x, w - 1 - x) < caveOceanW) continue;
390	    const sy = surface[x];
391	    for (let y = Math.floor(sy) + 6; y < h - 3; y++) {
392	      const depth = (y - sy) / (h - sy);
393	      const threshold = 0.30 - depth * 0.06; // 越深洞越多
394	      const v = fbm(noise2D, x * 0.028, y * 0.028, 3);
395	      if (v > threshold) store.setTileSilent(x, y, 0);
396	    }
397	  }
398	  // 2) 蠕虫隧道（连通大洞穴）
399	  const worms = 16 + Math.floor(rng.range(0, 8));
400	  for (let i = 0; i < worms; i++) {
401	    let wx = rng.range(w * 0.08, w * 0.92);
402	    let wy = rng.range(h * 0.35, h * 0.92);
403	    let angle = rng.range(0, Math.PI * 2);
404	    const len = rng.int(60, 200);
405	    for (let s = 0; s < len; s++) {
406	      angle += (noise2D(s * 0.05, i * 10) * 0.6 + rng.range(-0.3, 0.3));
407	      wx += Math.cos(angle) * 1.2;
408	      wy += Math.sin(angle) * 0.8;
409	      if (wx < 4 || wx > w - 5 || wy < h * 0.25 || wy > h - 6) break;
410	      const r = 2.2 + noise2D(s * 0.03, i * 5) * 1.8;
411	      carveCircle(store, wx, wy, Math.max(1.2, r));
412	    }
413	  }
414	}

416	function carveCircle(store: GenCtx['store'], cx: number, cy: number, r: number) {
417	  const x0 = Math.floor(cx - r), x1 = Math.ceil(cx + r);
418	  const y0 = Math.floor(cy - r), y1 = Math.ceil(cy + r);
419	  for (let x = x0; x <= x1; x++) {
420	    for (let y = y0; y <= y1; y++) {
421	      if (!store.inBounds(x, y)) continue;
422	      const dx = x - cx, dy = y - cy;
423	      if (dx * dx + dy * dy <= r * r) store.setTileSilent(x, y, 0);
424	    }
425	  }
426	}

428	function fbm(n2: (x: number, y: number) => number, x: number, y: number, oct: number): number {
429	  let v = 0, amp = 1, freq = 1, norm = 0;
430	  for (let i = 0; i < oct; i++) {
431	    v += n2(x * freq, y * freq) * amp;
432	    norm += amp;
433	    amp *= 0.5; freq *= 2;
434	  }
435	  return v / norm; // -1..1
436	}

441	// 3) 大灰烬丘陵(addTile 填充虚空)+ 横向灰烬团 + 挖掘隧道(填岩浆);
442	// 4) 底部 h-145/144 岩浆层 + 地狱石 58 矿脉
443	function hellPass({ store, rng }: GenCtx) {
444	  const st = store;
445	  const ASH = T.ASH;
446	  const HELLSTONE = TILE_BY_KEY['ore_hellstone']!;
447	  // 1) 边界 + 灰烬带 + 界下清空
448	  let boundary = st.h - rng.int(150, 189);
449	  for (let x = 0; x < st.w; x++) {
450	    boundary += rng.int(-3, 3);
451	    if (boundary < st.h - 190) boundary = st.h - 190;
452	    if (boundary > st.h - 160) boundary = st.h - 160;
453	    const ashTop = boundary - 20 - rng.int(0, 2);
454	    for (let y = ashTop; y < st.h; y++) {
455	      const i = st.idx(x, y);
456	      if (y >= boundary) {
457	        st.type[i] = 0; st.flags[i] = 0;
458	        st.liquid[i] = 0; st.liquidType[i] = 0;
459	      } else {
460	        st.type[i] = ASH; st.flags[i] = 1;
461	        st.wall[i] = 1;
462	      }
463	    }
464	  }
465	  // 2) 岩浆线
466	  let lavaY = st.h - rng.int(40, 69);
467	  for (let x = 10; x < st.w - 10; x++) {
468	    lavaY += rng.int(-10, 10);
469	    if (lavaY > st.h - 60) lavaY = st.h - 60;
470	    if (lavaY < st.h - 100) lavaY = st.h - 120;
471	    for (let y = lavaY; y < st.h - 10; y++) {
472	      const i = st.idx(x, y);
473	      if (!st.flags[i]) { st.liquid[i] = 255; st.liquidType[i] = 2; }
474	    }
475	  }
476	  const gs = ctxGs[0]!;
477	  // 3a) 大灰烬丘陵(1/50 列):巨型竖直灰烬 runner 填充虚空
478	  for (let x = 0; x < st.w; x++) {
479	    if (rng.next() * 50 >= 1) continue;
480	    let y = st.h - 65;
481	    while (!st.flags[st.idx(x, y)] && y > st.h - 135) y--;
482	    tileRunnerHell(st, rng, gs, rng.int(0, st.w - 1), y + rng.int(20, 49),
483	      rng.int(15, 19), 1000, ASH, true, 0, rng.int(1, 2), true);
484	  }
485	  // 3b) 每列 1/13:竖直灰烬柱 + 横向灰烬团 + 挖掘隧道(填岩浆)
486	  for (let x = 0; x < st.w; x++) {
487	    if (rng.next() * 13 >= 1) continue;
488	    let y = st.h - 65;
489	    while ((st.liquid[st.idx(x, y)] > 0 || st.flags[st.idx(x, y)]) && y > st.h - 140) y--;
490	    let scaleF = rng.int(1, 2);
491	    if (rng.next() * 3 < 1) scaleF *= 0.5;
492	    tileRunnerHell(st, rng, gs, x, y - rng.int(2, 4), rng.int(5, 29), 1000, ASH, true, 0, rng.int(1, 2), true);
493	    if (rng.next() < 0.5) {
494	      tileRunnerHell(st, rng, gs, x, y - rng.int(2, 4), Math.floor(rng.int(5, 14) * scaleF), Math.floor(rng.int(10, 14) * scaleF), ASH, true, 1, 0.3, false);
495	    }
496	    if (rng.next() < 0.5) {
497	      const s2 = rng.int(1, 2);
498	      tileRunnerHell(st, rng, gs, x, y - rng.int(2, 4), Math.floor(rng.int(5, 14) * s2), Math.floor(rng.int(10, 14) * s2), ASH, true, -1, 0.3, false);
499	    }
500	    tileRunnerHell(st, rng, gs, x + rng.int(-10, 9), y + rng.int(-10, 10), rng.int(5, 14), rng.int(5, 9), -2, false, rng.int(-1, 2), rng.int(-1, 2), false);
501	    if (rng.next() * 3 < 1) {
502	      tileRunnerHell(st, rng, gs, x + rng.int(-10, 9), y + rng.int(-10, 10), rng.int(10, 29), rng.int(10, 19), -2, false, rng.int(-1, 2), rng.int(-1, 2), false);
503	    }
504	    if (rng.next() * 5 < 1) {
505	      tileRunnerHell(st, rng, gs, x + rng.int(-15, 14), y + rng.int(-15, 9), rng.int(15, 29), rng.int(5, 19), -2, false, rng.int(-1, 2), rng.int(-1, 2), false);
506	    }
507	  }
508	  // 3c) 随机小挖掘(w 个)
509	  for (let n = 0; n < st.w; n++) {
510	    tileRunnerHell(st, rng, gs, rng.int(20, st.w - 21), rng.int(st.h - 180, st.h - 11),
511	      rng.int(2, 6), rng.int(2, 6), -2, false, 0, 0, false);
512	  }
513	  // 4) 底部岩浆层(h-145/144)+ 地狱石脉
514	  for (let x = 0; x < st.w; x++) {
515	    for (const dy of [145, 144]) {
516	      const i = st.idx(x, st.h - dy);
517	      if (!st.flags[i]) { st.liquid[i] = 255; st.liquidType[i] = 2; }
518	    }
519	  }
520	  for (let n = 0; n < st.w * st.h * 0.0008; n++) {
521	    tileRunnerHell(st, rng, gs, rng.int(0, st.w - 1), rng.int(st.h - 140, st.h - 1),
522	      rng.int(2, 6), rng.int(3, 6), HELLSTONE, false, 0, 0, false);
523	  }
524	}

540	    let removed = 0;
541	    for (let x = 2; x < st.w - 2; x++) {
542	      const surfY = Math.floor(surface[x]);
543	      for (let y = surfY + 2; y < st.h - 2; y++) {

545	        if (t !== T.DIRT && t !== T.STONE && t !== T.GRASS) continue;
546	        const below = st.isSolid(x, y + 1) || st.isSolid(x - 1, y + 1) || st.isSolid(x + 1, y + 1);

551	          removed++;
552	        }
553	      }
554	    }
555	    if (removed === 0) break;

560	function potPass({ store, surface, rng }: GenCtx) {
561	  const { w, h } = store;
562	  const POT = TILE_BY_KEY['pot']!;
563	  // 尝试次数随世界宽度；洞穴/浅层洞的地面（下方实心、上方 2x2 净空）放 1-3 个一组
564	  const tries = Math.floor(w / 6);
565	  for (let i = 0; i < tries; i++) {
566	    const x = rng.int(4, w - 5);
567	    const yMin = Math.floor(surface[x]) + 8;
568	    const yMax = Math.min(h - 6, Math.floor(surface[x]) + Math.floor(h * 0.45));
569	    if (yMax <= yMin) continue;
570	    const y = rng.int(yMin, yMax);
571	    // 2x2 净空 + 底部实心 = 洞内地面
572	    let ok = true;
573	    for (let dx = 0; dx < 2 && ok; dx++) {
574	      for (let dy = 1; dy <= 2; dy++) {
575	        if (store.isActive(x + dx, y - dy)) { ok = false; break; }
576	      }
577	    }
578	    if (!ok || !store.isSolid(x, y) || !store.isSolid(x + 1, y)) continue;
579	    // 该点 12 格内有其他瓦罐则跳过（避免扎堆）
580	    let nearby = false;
581	    for (let dx = -6; dx <= 6 && !nearby; dx++) {
582	      for (let dy = -4; dy <= 2; dy++) {
583	        if (store.get(x + dx, y + dy) === POT) { nearby = true; break; }
584	      }
585	    }
586	    if (nearby) continue;
587	    // 放 1-3 个（间距 2-3 格，各自需独立净空）
588	    const n = rng.int(1, 3);
589	    let px = x;
590	    for (let k = 0; k < n; k++) {
591	      if (store.get(px, y - 1) === 0 && store.get(px + 1, y - 1) === 0 &&
592	          store.get(px, y - 2) === 0 && store.get(px + 1, y - 2) === 0 && store.isSolid(px, y)) {
593	        store.setTileSilent(px, y - 1, POT, 0, 0);
594	        store.setTileSilent(px + 1, y - 1, POT, 18, 0);
595	        store.setTileSilent(px, y - 2, POT, 0, 18);
596	        store.setTileSilent(px + 1, y - 2, POT, 18, 18);
597	      }
598	      px += rng.int(2, 4);
599	      if (px > w - 4) break;
600	    }
601	  }
602	}

646	function liquidPass({ store, surface, rng }: GenCtx) {
647	  const { w, h } = store;
648	  // 地表洼地灌水（边缘海洋区已有水，跳过避免重复判定）
649	  const lakeOceanW = Math.max(24, Math.floor(w * 0.06)) + 6;
650	  let x = lakeOceanW;
651	  while (x < w - lakeOceanW) {
652	    const sy = surface[x];
653	    // 局部极小值
654	    if (sy > surface[x - 3] + 2 && sy > surface[x + 3] + 2) {
655	      const lakeW = rng.int(10, 26);
656	      // 找洼地范围内的最高岸
657	      let basinFloor = sy;
658	      for (let dx = 0; dx < lakeW; dx++) basinFloor = Math.max(basinFloor, surface[Math.min(w - 1, x + dx)]);
659	      const waterTop = Math.max(sy - 2, basinFloor - rng.int(2, 5));
660	      for (let dx = 0; dx < lakeW && x + dx < w; dx++) {
661	        // 逐列只填到该列自己的地表之上（waterTop < y < 该列 surface），
662	        // 绝不灌进实心地面内部
663	        const colFloor = surface[x + dx];
664	        for (let y = Math.floor(waterTop); y < Math.floor(colFloor); y++) {
665	          if (!store.inBounds(x + dx, y)) continue;
666	          // 只清非实心内容（杂草等），不动实心块
667	          const t = store.get(x + dx, y);
668	          if (t !== 0 && !TILE_DEFS[t]?.solid) store.setTileSilent(x + dx, y, 0);
669	          if (t === 0 || !TILE_DEFS[t]?.solid) store.setLiquid(x + dx, y, 255, 1);
670	        }
671	      }
672	      x += lakeW + 6;
673	    } else {
674	      x++;
675	    }
676	  }
677	  // 地下洞穴低处静水
678	  for (let i = 0; i < 60; i++) {
679	    const lx = rng.int(10, w - 10);
680	    const ly = rng.int(Math.floor(h * 0.5), h - 30);
681	    const poolW = rng.int(5, 14), poolH = rng.int(2, 4);
682	    for (let dx = 0; dx < poolW; dx++) {
683	      for (let dy = 0; dy < poolH; dy++) {
684	        const px = lx + dx, py = ly + dy;
685	        if (!store.inBounds(px, py)) continue;
686	        if (!store.isActive(px, py) && store.isSolid(px, py + 1)) {
687	          store.setLiquid(px, py, 255, 1);
688	        }
689	      }
690	    }
691	  }
692	}

695	function surfacePass({ store, surface, rng, world }: GenCtx) {
696	  const { w } = store;
697	  for (let x = 0; x < w; x++) {
698	    const sy = Math.floor(surface[x]);
699	    if (store.get(x, sy) === T.DIRT) {
700	      store.setTileSilent(x, sy, T.GRASS);
701	    }
702	    // 装饰：草丛/花/蘑菇——放在草块上方空气格（sy 是首个实心行，装饰在 sy-1）
703	    // 帧为 Tiles_3 表的像素偏移（TEdit frames 数据）：草丛 6 变体 / 花 9 色 / 蘑菇 (144,0)
704	    if (sy - 1 >= 0 && !store.isActive(x, sy - 1) && store.isSolid(x, sy)) {
705	      const GRASS_FX = [0, 18, 36, 54, 72, 90];
706	      const FLOWER_FX = [108, 126, 162, 180, 216, 252, 342, 540, 594];
707	      const r = rng.next();
708	      if (r < 0.34) store.setTileSilent(x, sy - 1, T.TALLGRASS, GRASS_FX[Math.floor(rng.next() * GRASS_FX.length) % 6], 0);
709	      else if (r < 0.48) {
710	        store.setTileSilent(x, sy - 1, T.FLOWER, FLOWER_FX[Math.floor(rng.next() * FLOWER_FX.length) % 9], 0);
711	        // 花簇：50% 概率向两侧蔓延 1 格（同簇同帧）
712	        const fx = store.frameX[store.idx(x, sy - 1)];
713	        for (const dx of [-1, 1]) {
714	          if (rng.next() < 0.5 && store.get(x + dx, sy - 1) === 0 && store.isSolid(x + dx, sy)) {
715	            store.setTileSilent(x + dx, sy - 1, T.FLOWER, fx, 0);
716	          }
717	        }
718	      }
719	      else if (r < 0.51) store.setTileSilent(x, sy - 1, T.MUSHROOM, 144, 0);
720	    }
721	  }
722	  // 树：正常间距 6-14；连续 24 格无树则放宽条件强制补种，保证顶部林区连续
723	  let x = rng.int(6, 14);
724	  let sinceLast = 0;
725	  while (x < w - 6) {
726	    const sy = Math.floor(surface[x]);
727	    const relaxed = sinceLast >= 24;
728	    // 软装饰（草丛/花/蘑菇/树苗）不挡树：基座格判定忽略，种前清掉
729	    const isSoft = (t: number) => t === T.TALLGRASS || t === T.FLOWER || t === T.MUSHROOM || t === T.SAPLING;
730	    // 不种在水里：树干基座格（地面上一格）必须无液体
731	    if (store.get(x, sy) === T.GRASS && (!store.isActive(x, sy - 1) || isSoft(store.get(x, sy - 1))) && store.liquid[store.idx(x, sy - 1)] === 0) {
732	      const th = rng.int(7, 14);
733	      let ok = true;
734	      for (let dy = 1; dy <= th; dy++) {
735	        if (store.isActive(x, sy - dy) && !(dy === 1 && isSoft(store.get(x, sy - dy)))) { ok = false; break; }
736	      }
737	      // 树冠 80×80 外扩 2 格：要求 ±2 列上方净空、地面实心；放宽时只查本列±1
738	      const range = relaxed ? 1 : 2;
739	      for (let dx = -range; dx <= range; dx++) {
740	        if ((store.isActive(x + dx, sy - 1) && !isSoft(store.get(x + dx, sy - 1))) || !(store.get(x + dx, sy) === T.GRASS || store.isSolid(x + dx, sy))) {
741	          ok = false; break;
742	        }
743	      }
744	      if (ok) {
745	        if (isSoft(store.get(x, sy - 1))) store.setTile(x, sy - 1, 0); // 清基座装饰
746	        // 原版 Tiles_5 22px 网格帧：树顶标记 (22,198) + 树干身 (0,0)
747	        for (let dy = 1; dy <= th; dy++) {
748	          store.setTileSilent(x, sy - dy, T.TREE, dy === th ? 22 : 0, dy === th ? 198 : 0);
749	        }
750	        // 侧枝：原版标记帧（44=左枝 / 66=右枝，y=198），干高 2..h-3 段两侧各 40%，
751	        // 每棵上限 2-4 枝（Tree_Branches 贴图由渲染端按标记帧绘制）
752	        const maxBranches = rng.int(2, 4);
753	        let branchCount = 0;
754	        for (let dy = 2; dy <= th - 3 && branchCount < maxBranches; dy++) {
755	          for (const side of [-1, 1]) {
756	            if (branchCount >= maxBranches) break;
757	            if (rng.int(0, 99) >= 40) continue;
758	            const bx = x + side, by = sy - dy;
759	            const bt = store.get(bx, by);
760	            const soft = bt === T.TALLGRASS || bt === T.FLOWER || bt === T.MUSHROOM || bt === T.SAPLING;
761	            if ((!store.isActive(bx, by) || soft) && store.get(bx, by + 1) !== T.TREE) {
762	              store.setTileSilent(bx, by, T.TREE, side < 0 ? 44 : 66, 198);
763	              branchCount++;
764	            }
765	          }
766	        }
767	        world.registerTree(x, sy - th, th);
768	        x += rng.int(6, 14);
769	        sinceLast = 0;
770	        continue;
771	      }
772	    }
773	    x++;
774	    sinceLast++;
775	  }
776	}

## 3. RNG 纪律

- **`Math.random()` 泄漏：仅 1 处**——`SeedEasterEggs.ts` L55-57（随机世界名拼接）。它不经过 `ctx.rng`，不污染 genRand 流，但严格说属于生成期用非种子随机；若要求"同种子 → 同世界名"，应改用种子派生。
- 所有 pass 共享单一 `ctx.rng`（`new RNG(hashString(seed))`）✓，无每 pass 自建 RNG。
- `RNG.int(min,max)` 闭区间 == C# `Next(min,max+1)` ✓；`rng.next() < 0.5` == `Next(2)==0` ✓；`rng.int(-10,10)*0.1` == `Next(-10,11)*0.1f` ✓。映射纪律良好。
- **与原版种子不复现**（内部可复现）：缺失 pass 不消费 RNG、自研 pass（ore/liquid/pot/surface/structure）按自己的顺序消费、pass 1 的 oreTiers 与 orePass 的 pair() 双重掷骰。当前契约只是"我们自己的流一致"，不是原版流——如果目标包含原版种子复现，需要按 105 pass 顺序补齐或补 RNG 占位消耗。

## 4. TileRunner.ts vs 原版 WorldGen.TileRunner（L46405）

**一致的部分（核对过）**：签名默认值（overRide=true/addTile=false/noYChange=false/ignoreTileType=-1）、初始速度掷骰与覆写条件、`type==59` 钻出顶即停、半径衰减 `strength*(num2/steps)`、边界钳 [1, w-1]、L1 距离 + 1.5%/1% 抖动、mudWall 深浅墙 15/64 分带与避让邻墙、granite/marble 0.3R 同步墙 180/178（单次 RNG，等价）、type<0 挖除（沙 53 保护、-2 液体填充条件 `<waterLine || >lavaLine`）、flag3 全套规则（宝石族 STONE_PLACED、11 种 NOT_CLEARABLE、45/147/189/190/196/460 无条件保护、沙上泥/粘土/地下沙漠规则、大理石花岗岩上泥、396/397 砂岩硬化沙矿石覆写）、noYChange 地表填墙 2、泥清水、**50..900 十二档加速阶梯**（含 Y 先 X 后的 RNG 次序）、每步抖动与 ±1 钳制（含 `!noYChange && type!=59 && num1<3` 只钳不抖分支）。

**差异**：
1. **缺失泥土垂直导向块（最重要）**——原版循环尾部的 `type==59 && !noYChange` 三条规则没移植：`vy` 钳 ±0.5、`py < rockLayer+100 → vy=1`、`py > maxTilesY-300 → vy=-1`。这是丛林泥 runner 被压进丛林带的原因；缺失后巨型泥 runner 会更自由地漂出丛林。建议补上。
2. `lava` 旗标：原版 -2 挖除的液体类型 = `lava(getGoodWorld && Next(4)==0) || y>lavaLine`，且海滩 ±(beachDistance+50) 内强制清 lava；我们简化为 `y>lavaLine ? 岩浆 : 水`。仅在 for-the-worthy 种子有视觉差，普通世界等价。
3. drunkWorld/getGoodWorld 开头的 strength/steps 修正与每 30 步瞬移：未实现（特殊种子整体未支持，一致性 OK）。
4. mudWall 邻墙检查：原版查 4 个方向（上/下/左/下——原版自身有把 `index2+1` 查两遍、漏 `index1+1` 的 bug），我们查 3 个（上/下/左）。行为接近原版"带 bug 的版本"，但未复刻那个重复检查；另外原版走 `PlaceWall(...,true)` 有合并语义，我们是直写。
5. 边界取整：原版 `(int)` 截断 vs 我们 `Math.floor`——钳到 1 之后实际无差。
6. 签名形态：options 对象替代 11 个位置参数（可读性选择，无行为差）；另加 `mudWall` 开关对应原版全局 `WorldGen.mudWall`（原版是全局旗标，我们是参数——若未来多 pass 并发需注意语义等价）。

附带：`digTunnel`（L48132）已同步移植，半径 [0.6×size, 2×size] 波动 + ±0.2 噪声游走 + 0.6 步长，是洞穴"有机感"的来源，实现正确。
