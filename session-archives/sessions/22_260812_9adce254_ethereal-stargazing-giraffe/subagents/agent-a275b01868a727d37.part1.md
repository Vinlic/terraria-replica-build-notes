async function warmZip(part, base, totalAll) {
  if (lastWarm.running) return;                  // 与 legacy warm 共用重入守卫(看门狗补发不叠跑)
  warmAbort = false;
  lastWarm = { done: base, total: totalAll, failed: 0, running: true };
  const cache = await getCache();

  // ① 下载片:单文件语义重试 ×3(300/600ms),与 legacy L145-155 同参
  let buf = null;
  for (let a = 0; a < 3 && !buf && !warmAbort; a++) {

          if (r.ok) buf = await r.arrayBuffer(); } catch {}

  }
  if (!buf) return zipDone(base, totalAll, part.files, /*下载失败=*/part.files);

  const have = new Set((await cache.keys()).map((r) => new URL(r.url).pathname.replace(/^\//, '')));
  const todo = part.entries.filter((n) => !have.has(n));

  // ③ 逐文件解压落缓存:unzipSync 单文件 filter —— 峰值内存 = zip 缓冲(64MB)+ 单文件;
  //    fflate 异步 API 依赖 new Worker,SW 内不可用,只能 Sync 族。

  //    换取扁平内存与逐文件可中断/可跳过)
  let done = base + (part.files - todo.length), failed = 0, put = 0;
  for (const name of todo) {
    if (warmAbort) break;                        // warm-cancel 即刻生效(每次 put 间都回事件循环)
    let data = null;
    try { data = fflate.unzipSync(buf, { filter: (f) => f.name === name })[name]; } catch {}
    if (data) {
      try { await cache.put(name, new Response(data, { headers: { 'Content-Type': mimeOf(name) } })); done++; }
      catch { failed++; }
    } else failed++;                             // 单文件 CRC 坏不炸整片
    if (++put % 200 === 0) await sleep(60);      // breath:磁盘落盘排空(沿用 BREATH 思路)
    if ((done & 63) === 0) postProgress('zip', done, totalAll, failed);
  }
  zipDone(done, totalAll, failed, warmAbort);

function zipDone(done, total, failed, aborted) {
  lastWarm = { done, total, failed, running: false };
  postMessageToPages({ type: 'warm-done', tag: 'zip', done, total, failed, aborted, conc: 1 });
}
