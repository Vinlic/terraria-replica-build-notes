function DrawWaterfall(style: number, alpha: number) {          // L211
  tileSolid[546] = false;                                       // L213
  for (const wf of waterfalls.slice(0, currentMax)) {           // L224
    let type = wf.type;
    if (type === 0) type = style;                               // L239-241
    let frameX = (type === 1 || type === 14) ? 32 * slowFrame   // L246
                                          : 32 * regularFrame;  // L353
    let x = wf.x, y = wf.y;
    let dx = 0, dy = 0, turnCount = 0, lastDx = 0, prevDy = 0;
    let yOff = 0;                                               // num11
    const maxLen = waterfallDist;                               // L357, 788
    for (let step = 0; step < maxLen; step++) {                 // L359
      if (turnCount < 2) addLightForType(type, x, y);           // L363-420
      const t = tile(x, y);
      if (isSolidBlock(t)) break;                               // L427
      const below = tile(x, y+1), left = tile(x-1,y), right = tile(x+1,y);
      // 走向决策 → (dx, dy, yOff), turnCount++ 计偏折          // L452-502
      // 流经 160/262-268 时改写 type (彩虹/荧光)               // L516-530
      let light = Lighting.getColor(x, y);                      // L535
      let a = (type===1) ? 1 : (type===14) ? 0.8                // L538-548
             : (t.wall!==0 || y>=worldSurface ? 0.6*alpha : alpha);
      if (step > maxLen-10) a *= (maxLen-step)/10;              // L550-551
      const col = applyTypeColorRules(light, a, type);          // L552-586
      drawTile(waterfallTex[type], x*16, y*16+yOff, frameX, 0, 16, 16 - t.liquid/16, col);
      // 汇流/转角时额外画 (frameX,24,32,…) 的 2 格宽拼接块     // L642-708
      // 落底时画 2px 斜切片溅花                                // L719-771
      if (t.liquid > 0 && !t.halfBrick()) break;                // L777-778
      x += dx; y += dy;                                         // L782-783
    }
  }
  tileSolid[546] = true;                                        // L802
}
function Draw() {                                               // L805
  for (const wf of waterfalls) wf.stopAtStep = waterfallDist;   // L807
  Main.drewLava = false;
  for (const [liqIdx, style] of STYLE_MAP)                      // L810-832
    if (liquidAlpha[liqIdx] > 0) DrawWaterfall(style, liquidAlpha[liqIdx]);
}
