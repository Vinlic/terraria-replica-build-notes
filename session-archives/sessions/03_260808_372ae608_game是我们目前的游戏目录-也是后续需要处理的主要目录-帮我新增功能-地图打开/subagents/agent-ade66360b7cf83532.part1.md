322	  const base = { ChanceNumerator: 1, ChanceDenominator: 1, MinimumItemDropsCount: 1, MaximumItemDropsCount: 1, MinimumStackPerChunkBase: 1, MaximumStackPerChunkBase: 1, BonusMinDropsPerChunkPerPlayer: 0, BonusMaxDropsPerChunkPerPlayer: 0 };

331	function resolveRule(node) {
332	  // varRef 解引用 + 参数对象实化
333	  const walk = (r) => {
334	    if (!r || typeof r !== 'object') return r;
335	    if (r.k === 'varRef') return walk(rules.get(r.name) ?? { k: 'unknown', raw: `var:${r.name}` });
336	    for (const key of Object.keys(r)) {
337	      if (key === 'chain') { r[key] = r[key].map(c => ({ ...c, rule: walk(c.rule) })); continue; }
338	      const v = r[key];
339	      if (v && typeof v === 'object' && !Array.isArray(v) && v.k) r[key] = walk({ ...v });
340	      else if (Array.isArray(v)) r[key] = v.map(x => (x && typeof x === 'object' && x.k ? walk({ ...x }) : x));
341	    }
342	    return r;
343	  };
344	  const out = walk({ ...node });
345	  // 参数对象实化（dropOneByOne.params / oneOfRules 内不含）
346	  if (out.k === 'dropOneByOne') {
347	    const pn = out.params;
348	    out.params = pn?.t === 'id' ? (paramss.get(pn.name) ?? null)
349	      : pn?.t === 'paren' ? paramsObj(pn.inner) : paramsObj(pn);
350	  }
351	  return out;
352	}
353	function emit(npcs, ruleNode, global = false) {

358	/** 注册语句尾部的 .OnXxx(...) 链挂到规则上（gate.OnSuccess(r) → gate.chain） */
359	function attachChains(rule, chains) {
360	  for (const c of chains) {
361	    if (!rule || rule.k === 'unknown') return;
362	    if (rule.k === 'gate' && c.fn === 'OnSuccess') {
363	      // gate 的链语义：OnSuccess 挂 chain（LeadingConditionRule 门卫放行链）
364	      rule.chain = rule.chain ?? [];
365	      rule.chain.push({ on: 'success', rule: c.rule });
366	    } else {
367	      rule.chain = rule.chain ?? [];
368	      rule.chain.push({ on: CHAIN_FN[c.fn] ?? 'success', rule: c.rule });
369	    }
370	  }

386	/**
387	 * 统一注册处理。expr = 完整 postfix（可含链），rootCall = 剥链后的注册调用，
388	 * refVar = 可选的变量名（IItemDropRule rule = RegisterTo...(…) 时记录活动引用，
389	 * 之后的 rule.OnSuccess(...) 语句直接挂到已 emit 的规则上）
390	 */
391	function handleRegistration(s, expr, rootCall, refVar) {
392	  const reg = rootCall.fn;
393	  const a = rootCall.args;
394	  // 链（expr 与 rootCall 之间的层）
395	  const chains = [];
396	  if (expr !== rootCall) {
397	    let n = expr;
398	    while (n !== rootCall) {
399	      chains.unshift({ fn: n.fn, rule: toRule(n.args[1]) });
400	      n = n.args[0];
401	    }
402	  }
403	  if (reg === 'RegisterToGlobal') {
404	    const r = toRule(a[0]);
405	    if (chains.length) attachChains(r, chains);
406	    emit([], r, true);
407	    if (refVar) { emittedRefs.set(refVar, { rule: r, entry: null }); rules.set(refVar, r); }
408	    return;
409	  }
410	  if (reg === 'RegisterToNPC') {
411	    let id = null;
412	    if (a[0].t === 'num') id = [a[0].v];
413	    else if (a[0].t === 'id' && ints.has(a[0].name)) id = [ints.get(a[0].name)];
414	    if (id == null) return;  // npcNetIds[i] 变体拷贝辅助
415	    const r = toRule(a[1]);
416	    if (chains.length) attachChains(r, chains);
417	    emit(id, r);
418	    if (refVar) { emittedRefs.set(refVar, { rule: r, entry: entries[entries.length - 1] }); rules.set(refVar, r); }
419	    return;
420	  }
421	  // Multiple 系
422	  const rule = toRule(a[0]);
423	  if (chains.length) attachChains(rule, chains);
424	  const ids = [];
425	  for (const arg of a.slice(1)) {
426	    if (arg.t === 'num') ids.push(arg.v);
427	    else if (arg.t === 'id' && arrays.has(arg.name)) ids.push(...arrays.get(arg.name));
428	    else if (arg.t === 'newarr') ids.push(...arg.items.map(x => lit(x)).filter(v => v !== undefined));
429	  }
430	  if (ids.length) { emit(ids, rule); if (refVar) { emittedRefs.set(refVar, { rule, entry: entries[entries.length - 1] }); rules.set(refVar, rule); } }
431	  else stats.unknown.push(`regids:${render(a[0])}`);
432	}

437	  const s = stmt.trim();

452	    m = /^(?:int|short|long)\s+(\w+)\s*=\s*(-?\d+)/.exec(s);

456	    if (m) {
457	      const expr = parseExpr(s, s.indexOf('=') + 1);
458	      paramss.set(m[1], paramsObj(expr));
459	      return;
460	    }
461	    // Parameters 拷贝+变更：p2 = p; p2.Prop = N;
462	    m = /^DropOneByOne\.Parameters\s+(\w+)\s*=\s*(\w+)\s*;/.exec(s);
463	    if (m) { const src = paramss.get(m[2]); if (src) paramss.set(m[1], { ...src }); return; }

468	    if (m) {
469	      const rhs = s.slice(s.indexOf('=') + 1).replace(/;$/, '');
470	      const isCond = s.startsWith('IItemDropRuleCondition') || s.startsWith('Conditions.');
471	      const expr = parseExpr(rhs, 0);
472	      if (isCond) rules.set(m[1], { k: '__cond', cond: condOf(expr) ?? condOfVar(expr) });
473	      else {

475	        const rootCall = stripChains(expr);
476	        if (rootCall.t === 'call' && rootCall.fn.startsWith('RegisterTo')) {
477	          handleRegistration(s, expr, rootCall, m[1]);
478	        } else {
479	          rules.set(m[1], toRule(expr));
480	        }
481	      }
482	      return;
483	    }
484	    // 变量上挂链 / Add：rule.OnSuccess(...) / rule2.OnFailedRoll(...)；
485	    // 变量可能是规则变量（rules）或已 emit 条目引用（emittedRefs，EoW 模式）
486	    m = /^(\w+)\.(OnSuccess|OnFailedRoll|OnFailedConditions)\(/.exec(s);
487	    if (m && (rules.has(m[1]) || emittedRefs.has(m[1]))) {
488	      const expr = parseExpr(s, s.indexOf(m[0]));
489	      // postfix 结构：call(fn=OnX, args=[varRef, rule])
490	      if (expr.t === 'call' && expr.base) {
491	        const ref = emittedRefs.get(m[1]);
492	        const host = ref ? ref.rule : rules.get(m[1]);
493	        host.chain = host.chain ?? [];
494	        host.chain.push({ on: CHAIN_FN[m[2]], rule: toRule(expr.args[1]) });
495	      }
496	      return;
497	    }
498	    // RemoveFromMultipleNPCs(entry, ids)：把 ids 从已 emit 条目的 npc 列表移除
499	    m = /^RemoveFromMultipleNPCs\((\w+),\s*(\w+)\)/.exec(s);
500	    if (m && emittedRefs.has(m[1])) {
501	      const ids = arrays.get(m[2]) ?? [];
502	      const entry = emittedRefs.get(m[1]).entry;
503	      if (entry) entry.npcs = entry.npcs.filter((n) => !ids.includes(n));
504	      return;
505	    }
506	    // 注册调用（含链式 .OnSuccess(...) 尾巴）：剥链到最内层注册调用
507	    for (const reg of ['RegisterToMultipleNPCsNotRemixSeed', 'RegisterToMultipleNPCsRemixSeed', 'RegisterToMultipleNPCs', 'RegisterToGlobal', 'RegisterToNPC']) {
508	      const idx = s.indexOf(reg + '(');
509	      if (idx !== 0 && !(idx > 0 && /\w/.test(s[idx - 1]) === false && idx <= 2)) continue;
510	      if (idx < 0) continue;
511	      const expr = parseExpr(s, idx);
512	      if (expr.t !== 'call') continue;
513	      const node = stripChains(expr);
514	      if (node.t !== 'call' || node.fn !== reg) continue;
515	      handleRegistration(s, expr, node, null);
516	      return;
517	    }
518	  } catch (e) {
519	    stats.unknown.push(`EXC:${s.slice(0, 70)} (${e.message})`);
520	  }
