36	  /** AI_006 头部（L18645 通用常数 maxSpd=8 accel=0.07；穿墙直行；段链跟随） */
37	  private wormAI(game: GameHooks, player: Player | null) {
38	    const maxSpd = 8, accel = 0.07;
39	    // 朝向：有玩家朝玩家，无玩家缓慢巡游
40	    let dx: number, dy: number;
41	    if (player) { dx = player.cx - this.cx; dy = player.cy - this.cy; }
42	    else { dx = Math.cos(this.aiT * 0.02) * 10; dy = Math.sin(this.aiT * 0.013) * 10; }
43	    const d = Math.hypot(dx, dy) || 1;
44	    this.vx += (dx / d) * accel;
45	    this.vy += (dy / d) * accel;
46	    const spd = Math.hypot(this.vx, this.vy);
47	    if (spd > maxSpd) { this.vx = (this.vx / spd) * maxSpd; this.vy = (this.vy / spd) * maxSpd; }
48	    this.facing = this.vx > 0 ? 1 : -1;
49	    // 蠕虫穿墙：直接位移（原版 noTileCollide）
50	    this.x += this.vx;
51	    this.y += this.vy;
52	    // 段链跟随：每段贴前一段的上一位置
53	    for (let s = this.wormNext; s; s = s.wormNext) {
54	      const fx = s.wormFollow!;
55	      s.x = fx.prevX;
56	      s.y = fx.prevY;
57	      s.facing = fx.facing;
58	    }
59	  }

63	    const segs: Enemy[] = [];
64	    const bodyId = head.vanillaId! + 1, tailId = head.vanillaId! + 2;
65	    let prev = head;

68	      const s = Enemy.fromVanilla(id, head.cx, head.cy);
69	      if (!s) continue;

71	      prev.wormNext = s;
72	      prev = s;
73	      segs.push(s);
74	    }
75	    return segs;

146	    this.prevX = this.x; this.prevY = this.y;
147	    this.aiT++;
148	    if (this.iframes > 0) this.iframes--;
149	    if (this.hpBarT > 0) this.hpBarT--;
150	    if (this.squash !== 0) this.squash *= 0.85;
151	    this.animT++;
152	    this.walkCycleT += Math.abs(this.vx); // 原版 zombie/fighter 行走帧按 |vx| 累加（L77072/L70216）
153	
154	    const player = (game as unknown as { player: Player }).player;
155	    const hasPlayer = !!player && !player.dead;
156	
157	    // ---- 原版数据驱动路径：按 aiStyle 分发到已移植的 AI 家族（分发后落入共享尾段） ----
158	    // 蠕虫身体段（wormFollow 非空）无 AI：位置由头部 wormAI 沿链驱动，但仍走共享尾段（接触伤害等）
159	    // 小动物（critter）优先走被动游荡，绝不落入 default 的追击 fallback
160	    if (this.vanilla && !this.wormFollow && this.vanilla.critter) {
161	      this.critterWanderAI(game, hasPlayer ? player : null);
162	    } else if (this.vanilla && !this.wormFollow) {
163	      const p = hasPlayer ? player : null;
164	      switch (this.vanilla.aiStyle) {
165	        case 1: this.slimeAI(game, p); break;                          // AI_001 史莱姆
166	        case 2: this.floatEyeAI(game, p); break;                        // AI_002 飘浮眼（原版核）
167	        case 3: this.fighterAI(game, p); break;                        // AI_003 战士族（原版通用核）
168	        case 5: this.swarmerAI(game, p); break;                         // AI_005 噬魂怪族（原版蜂群核）
169	        case 6: this.wormAI(game, p); break;                           // AI_006 蠕虫族（头）
170	        case 8: this.casterAI(game, p); break;                          // AI_008 法师族（传送+弹幕）
171	        case 14: this.batAI(game, p); break;                            // AI_014 蝙蝠/飞行族（1.4.5.6 原版核）
172	        case 16: this.swimAI(game, p); break;                           // AI_016 游泳族（鲨鱼/食人鱼）
173	        case 18: this.jellyfishAI(game, p); break;                      // AI_018 水母族（脉冲游动）
174	        case 22: this.ghostAI(game, p); break;                          // AI_022 幽灵族（穿墙缓速）
175	        case 26: this.chargerAI(game, p); break;                        // AI_026 独角兽/冲锋族（原版核）
176	        case 107: this.chargerAI(game, p, 1); break;                    // AI_107 ImprovedWalkers（0.07/±1.0 walker 档，NPC.145.cs:43153 引擎核同构）
177	        default: this.zombieAI(game, p); break;                        // 其余家族待逐个移植
178	      }
179	    } else switch (this.key) {
180	      case 'slime_green':
181	      case 'slime_blue': this.slimeAI(game, hasPlayer ? player : null); break;
182	      case 'zombie': this.zombieAI(game, hasPlayer ? player : null); break;
183	      case 'demon_eye':
184	      case 'cave_bat': this.flyAI(game, hasPlayer ? player : null, 0.09); break;
185	      case 'servant_of_cthulhu': this.flyAI(game, hasPlayer ? player : null, 0.22); break;
186	      case 'eye_of_cthulhu': this.eocAI(game, hasPlayer ? player : null); break;

189	    // 入水溅落声（与主角一致）：音量随与主角距离衰减，Boss 更响
190	    const eli = game.world.store.liquid[game.world.store.idx(
191	      Math.floor(this.cx / TILE), Math.floor(this.cy / TILE),
192	    )];
193	    const wasWater = this.inWater;
194	    this.inWater = eli > 80;
195	    if (this.inWater && !wasWater && hasPlayer) {
196	      const d = Math.hypot(player.cx - this.cx, player.cy - this.cy);
197	      const vol = Math.max(0.12, Math.min(1, 1 - d / (TILE * 45))) * (this.def.boss ? 1.2 : 0.9);
198	      game.playSfx('splash', vol);

202	    // 否则 damagePlayer(0) 仍会击退/闪红，表现为"兔子撞我有攻击性"
203	    if (hasPlayer && this.aabbOverlaps(player) && this.def.damage > 0) {
204	      game.damagePlayer(this.def.damage, this.cx, this.cy, this);
205	    }
206	
207	    // 白天烧夜间怪（恶魔眼/僵尸在白天消失）
208	    if ((this.def.nightOnly) && game.world.clock.isDay && !this.def.boss) {
209	      if (game.world.clock.dayFactor > 0.85) this.dead = true;
210	    }
211	    // 距玩家过远清除
212	    if (hasPlayer) {
213	      const d = Math.hypot(player.cx - this.cx, player.cy - this.cy);
214	      if (!this.def.boss && d > TILE * 90) this.dead = true;
215	    }

218	  private groundPhysics(game: GameHooks, jumpV = 0, accel = 0.12, maxSpd = 1, buoyant = false) {
219	    // 浮力生物（史莱姆）：水中上浮 + 阻尼，漂在水面而不是沉底
220	    if (buoyant) {
221	      const st = game.world.store;
222	      const li = st.liquid[st.idx(Math.floor(this.cx / TILE), Math.floor(this.cy / TILE))];
223	      if (li > 80) {
224	        if (this.state === 1) {
225	          // 跳跃中：弱重力助其跃出水面（浮力阻尼会把起跳速度吃掉导致困在水里）
226	          this.vy = Math.min(this.vy + GRAVITY * 0.3, MAX_FALL_SPEED);
227	          moveAndCollide(this, game.world, this.vx, this.vy);
228	          this.vx *= 0.98;
229	          return;
230	        }
231	        this.vy -= 0.32;
232	        this.vy *= 0.92;
233	        this.vx *= 0.95;
234	        moveAndCollide(this, game.world, this.vx, this.vy);
235	        return;
236	      }
237	    }
238	    this.vy = Math.min(this.vy + GRAVITY, MAX_FALL_SPEED);
239	    this.vx = Math.max(-maxSpd, Math.min(maxSpd, this.vx + accel * Math.sign(this.vx || 1) * 0));
240	    if (jumpV && this.onGround) this.vy = -jumpV;
241	    moveAndCollide(this, game.world, this.vx, this.vy);
242	    this.vx *= this.onGround ? 0.8 : 0.98;
243	  }

245	  private slimeAI(game: GameHooks, player: Player | null) {
246	    // 水中漂浮状态：漂在水面时也可蓄力起跳
247	    const st = game.world.store;
248	    const inWater = st.liquid[st.idx(Math.floor(this.cx / TILE), Math.floor(this.cy / TILE))] > 80;
249	    const surfaced = inWater && Math.abs(this.vy) < 0.8; // 漂在水面附近（准地面）
250	    // 状态机：0 蓄力 → 1 跳跃
251	    if (this.state === 0) {
252	      this.vx *= 0.6;
253	      if (this.onGround) this.squash = Math.min(1, this.squash + 0.08);
254	      const canAct = this.onGround || surfaced;
255	      // 脱困冷却内（stuckCd>0）落地/浮面立刻再跳，不在原地等 50 tick 空转
256	      // 索敌：受击后（iframes 刚触发过）或玩家贴近 <6 格才追；否则随机方向游荡跳
257	      const hurt = this.iframes > 0;
258	      const close = !!player && Math.abs(player.cx - this.cx) < TILE * 6;
259	      if (canAct && (this.aiT % 50 === 0 || this.stuckCd > 0 || ((hurt || close) && this.aiT > 40))) {
260	        // 起跳方向：掉头冷却内用当前朝向（不再朝玩家死磕同一堵墙）
261	        let dir = this.facing;
262	        if (player && this.stuckCd === 0) {
263	          const dx = player.cx - this.cx;
264	          if (Math.abs(dx) > 8) dir = Math.sign(dx);
265	        }
266	        this.facing = dir;
267	        const dist = player ? Math.min(1, Math.abs(player.cx - this.cx) / (TILE * 10)) : 0.4;
268	        // 前方障碍高度探测（1-2 格）→ 按需起跳力度。
269	        // 普通跳 vy-4.2 在 GRAVITY=0.36 下脚底只升 24.5px（1.5 格）——
270	        // 精灵贴图放大 1.25 倍看着跳很高，但碰撞盒过不了 32px 的两格墙
271	        const fx = Math.floor((this.cx + dir * (this.w / 2 + 2)) / TILE);
272	        const fy = Math.floor((this.y + this.h - 1) / TILE);
273	        let obsH = 0;
274	        for (let k = 0; k < 3; k++) {
275	          if (st.isSolid(fx, fy - k)) obsH = k + 1; else break;
276	        }
277	        const headroom = obsH > 0 && obsH <= 2 && !st.isSolid(fx, fy - obsH) && !st.isSolid(fx, fy - obsH - 1);
278	        if (headroom) {
279	          // 障碍跳：1 格 -4.6（脚底升 29px）/ 2 格 -6.0（升 50px），慢速横移保证越过块面
280	          this.vx = dir * (obsH === 1 ? 1.6 : 1.2);
281	          this.vy = obsH === 1 ? -4.6 : -6.0;
282	        } else {
283	          this.vx = dir * (1.6 + dist * 2.6);
284	          // 水中起跳加力：跃出水面爬上岸（普通力度会被浮力吃掉困在水里）
285	          this.vy = inWater ? -(5.6 + dist * 1.6) : -(4.2 + dist * 2.4);
286	        }
287	        this.jumpStartX = this.cx;
288	        this.state = 1;
289	        this.squash = -0.7;
290	      }
291	    } else {
292	      // 空中转向（与主角空操一致）：跳跃中持续朝面向水平加速——
293	      // 没有它起跳瞬间 vx 被墙清零后就只剩原地垂直跳，永远出不去
294	      this.vx = Math.max(-2.6, Math.min(2.6, this.vx + this.facing * 0.12));
295	      const landed = this.onGround && this.vy >= 0 && this.aiT > 10;
296	      if (landed || (surfaced && this.aiT > 20)) {
297	        // 起跳后几乎没位移（被墙弹回）→ 直接记一次失败，快速累计触发翻面
298	        if (Math.abs(this.cx - this.jumpStartX) < 6) this.stuckT += 10;
299	        this.state = 0;
300	        this.aiT = 0;
301	      }
302	    }
303	    this.groundPhysics(game, 0, 0.12, 1, true);
304	    // 卡墙脱困（水中/陆地通用）：撞墙计数用"缓慢衰减"而非硬清零——
305	    // 史莱姆是"跳一下撞墙→落回水中漂离"的循环，撞墙帧之间 hitWall 为 false，
306	    // 硬重置会让跨跳跃的撞墙永远攒不满阈值。累计 24 → 翻面 + 逃亡冷却
307	    // （水中 240 tick：足够连续多跳横渡水域脱离，否则冷却一到又朝墙对面的玩家折返死磕）
308	    if (this.stuckCd > 0) this.stuckCd--;
309	    if (this.hitWall) {
310	      this.stuckT += 3;
311	    } else {
312	      this.stuckT = Math.max(0, this.stuckT - 1);
313	    }
314	    if (this.stuckT > 24) {
315	      this.stuckT = 0;
316	      this.stuckCd = inWater ? 240 : 90;
317	      this.facing = -this.facing;
318	    }
319	  }

321	  private zombieAI(game: GameHooks, player: Player | null) {
322	    // 卡墙脱困：朝玩家走却持续撞墙（高墙跳不过去）→ 掉头往另一侧探索
323	    if (this.onGround && this.hitWall && player) {
324	      if (++this.stuckT > 80) {
325	        this.stuckT = 0;
326	        this.stuckCd = 90; // 掉头冷却：期间不再朝玩家折返
327	      }
328	    } else if (!this.hitWall) {
329	      this.stuckT = 0;
330	    }
331	    if (this.stuckCd > 0) this.stuckCd--;
332	    let dir = this.facing;
333	    if (player && this.stuckCd === 0) {
334	      dir = Math.sign(player.cx - this.cx) || 1;
335	      // 跳过障碍
336	      if (this.onGround && this.hitWall) this.vy = -6;
337	      // 门：会尝试跳，不做开门
338	    } else if (this.hitWall && this.onGround) {
339	      dir = -this.facing; // 掉头期间撞墙就反向
340	    }
341	    this.facing = dir;
342	    this.vx += dir * 0.08;
343	    this.vx = Math.max(-0.9, Math.min(0.9, this.vx));
344	    this.groundPhysics(game, 0);
345	  }

347	  /** AI_003_Fighters 通用核（NPC.cs:24559-24733，剥离 per-type 特例）：
348	   *  前方 tile 探测 + 四级跳（2 格墙 -8 / 1 格墙 -7 / 卡半格 -5 / 前方悬空且目标在上 -8 加速），
349	   *  加速度 0.1、最大速度 1.0（原版 num1/num2 常量），门/高门交互待移植 */
350	  private fighterAI(game: GameHooks, player: Player | null) {
351	    const st = game.world.store;
352	    // 方向：朝玩家（无玩家则保持）
353	    if (player) this.facing = player.cx > this.cx ? 1 : -1;
354	    const dir = this.facing;
355	    // 前方探测点（L24561-24562）：体中心前 15px、脚底上 15px
356	    const fx = Math.floor((this.x + this.w / 2 + 15 * dir) / TILE);
