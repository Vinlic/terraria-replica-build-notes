25	function readVarint(u8: Uint8Array, pos: { i: number }): number {
26	  let v = 0, shift = 0;
27	  while (pos.i < u8.length) {
28	    const b = u8[pos.i++];
29	    v |= (b & 0x7f) << shift;
30	    if (!(b & 0x80)) break;
31	    shift += 7;
32	  }
33	  return v;
