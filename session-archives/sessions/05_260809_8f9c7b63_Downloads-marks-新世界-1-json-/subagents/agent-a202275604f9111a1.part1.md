100	/** 整图硬 alpha：alpha ≥128 → 255，<128 → 0（并清零 RGB），消除提取 PNG 的半透明镶边 */
101	function hardAlpha(img: HTMLImageElement): HTMLCanvasElement {
102	  const c = document.createElement('canvas');
103	  c.width = img.width; c.height = img.height;
104	  const ctx = c.getContext('2d')!;
105	  ctx.drawImage(img, 0, 0);
106	  const d = ctx.getImageData(0, 0, c.width, c.height);
107	  const px = d.data;
108	  for (let i = 0; i < px.length; i += 4) {
109	    if (px[i + 3] >= 128) px[i + 3] = 255;
110	    else {
111	      px[i] = 0; px[i + 1] = 0; px[i + 2] = 0; px[i + 3] = 0;
112	    }
113	  }
114	  ctx.putImageData(d, 0, 0);
115	  return c;
116	}

130	  async load(onProgress?: (p: number) => void): Promise<void> {
131	    const files = Object.keys(this.data.files);
132	    // 原版 vanilla 素材与 vanilla-ui 贴图全部不在启动预载(8550 请求/主菜单 2GB 根因):
133	    // 图块/墙/NPC 表 → Game.newWorld/loadWorld 里 preloadVanillaWorld() 预载
134	    // (onWorldReady 之前完成,首帧 chunk 烘焙无回退);物品图标 → vicon 按需
135	    // 懒加载 + 进世界后 prefetchIcons() 后台补齐;UI 贴图 → vui() 按需
136	    // 懒加载(全部 11 处消费方每帧重查,首帧 null 自兜底)
137	    const vfiles: string[] = [];
138	    const uifiles: string[] = [];
139	    let done = 0;
140	    const total = files.length + vfiles.length + uifiles.length;
141	    await Promise.all([
142	      ...files.map((f) => new Promise<void>((resolve) => {
143	        const img = new Image();
144	        img.onload = () => {
145	          // 根源处理：整图硬 alpha —— 抗锯齿半透明像素（提取 PNG 的灰/黑镶边来源）

147	          this.images.set(f, hardAlpha(img));
148	          done++;
149	          onProgress?.(done / total);
150	          resolve();
151	        };
152	        img.onerror = () => resolve();
153	        img.src = `sprites/${encodeURI(f)}`;
154	      })),
155	    ]);
156	    // 人工标注（可选，缺失时回退）
157	    try {
158	      const r = await fetch('sprites/annotations.json');
159	      if (r.ok) this.annotations = await r.json();
160	    } catch { /* 无标注 */ }
161	  }

285	    const list = [...new Set(files)];
286	    let done = 0;
287	    await Promise.all(list.map((f) => new Promise<void>((resolve) => {
288	      if (this.vimages.has(f)) { done++; onProgress?.(done / list.length); return resolve(); }
289	      const img = new Image();
290	      img.onload = () => {
291	        this.vimages.set(f, img);

293	        // 大表(2000px 级)首帧仍会卡一拍——"预载"语义要求字节+解码双就绪
294	        img.decode().then(() => { done++; onProgress?.(done / list.length); resolve(); }, () => resolve());
295	        this.onVImageLoaded?.(f);
296	      };
297	      img.onerror = () => resolve();
298	      img.src = `sprites/${encodeURI(f)}`;
299	    })));
300	  }

427	export function atlasIconForKey(atlas: SpriteAtlas, key: string): DrawRect | null {
428	  const vid = VANILLA_ITEM_ICON_MAP[key];
429	  if (vid !== undefined) return atlas.vicon(vid);
430	  // 阶段 5:vi_<id>_* 全量物品回退原版图标
431	  if (key.startsWith('vi_')) {
432	    const id = parseInt(key.slice(3), 10);
433	    if (Number.isFinite(id)) return atlas.vicon(id);
434	  }
435	  // 阶段 5:place_v_* 放置物品回退用图块贴图首帧作图标
436	  if (key.startsWith('place_v_')) {
437	    const tk = key.slice('place_'.length);
438	    const tid = TILE_BY_KEY[tk];
439	    if (tid !== undefined) {
440	      const td = TILE_DEFS[tid];
441	      // vframeAt 而非 vframe:窄条表(如压板 135 的 16×200 竖条)cols=0 会让 vframe 判越界
442	      if (td?.vanilla) return atlas.vframeAt(td.vanilla.sheet, 0, 0) ?? atlas.vframe(td.vanilla.sheet, 0, 0);
443	    }
444	  }
445	  return null;
446	}

## 4. preloadIcons 与 ensureVImage 的关系

- 调用方:`core/Game.ts:383`(`preloadSceneAssets` 内 `void a.preloadIcons()`,完成后 `onInventoryChanged` 刷新背包)+ `mainFlow.ts:86`(`prefetchIcons()`)
